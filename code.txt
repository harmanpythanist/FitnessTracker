        // ─────────────────────────────────────────────────────────────
        //  Arc Gauge + Mini Stat Cards
        // ─────────────────────────────────────────────────────────────
        void AddGoalProgressSection(int top, double todayTotal, double goal,
                                    int todayCount, double allTime)
        {
            int W = pnlMain.Width - SideMargin * 2;
            int H = 170;
            double pct = goal > 0 ? Math.Min(1.0, todayTotal / goal) : 0;
            bool achieved = todayTotal >= goal && goal > 0;
            Color arcColor = achieved
                ? Color.FromArgb(40, 167, 69)
                : Color.FromArgb(13, 110, 253);

            var card = WhiteCard(SideMargin, top, W, H);

            card.Paint += (s, e) =>
            {
                var g = e.Graphics;
                g.SmoothingMode = System.Drawing.Drawing2D.SmoothingMode.AntiAlias;

                using var border = new Pen(Color.FromArgb(218, 218, 224), 1);
                g.DrawRectangle(border, 0, 0, card.Width - 1, card.Height - 1);

                // ── Arc ───────────────────────────────────────────────
                int cx = 110, cy = 130, r = 80;
                float startAngle = 180f, fullSweep = 180f;
                float fillSweep = (float)(fullSweep * pct);

                using var trackPen = new Pen(Color.FromArgb(232, 232, 238), 15);
                trackPen.StartCap = System.Drawing.Drawing2D.LineCap.Round;
                trackPen.EndCap = System.Drawing.Drawing2D.LineCap.Round;
                g.DrawArc(trackPen, cx - r, cy - r, r * 2, r * 2, startAngle, fullSweep);

                if (fillSweep > 0)
                {
                    using var fillPen = new Pen(arcColor, 15);
                    fillPen.StartCap = System.Drawing.Drawing2D.LineCap.Round;
                    fillPen.EndCap = System.Drawing.Drawing2D.LineCap.Round;
                    g.DrawArc(fillPen, cx - r, cy - r, r * 2, r * 2, startAngle, fillSweep);

                    double tip = Math.PI + Math.PI * pct;
                    int tipX = (int)(cx + r * Math.Cos(tip));
                    int tipY = (int)(cy + r * Math.Sin(tip));
                    g.FillEllipse(Brushes.White, tipX - 8, tipY - 8, 16, 16);
                    using var db = new Pen(arcColor, 2.5f);
                    g.DrawEllipse(db, tipX - 8, tipY - 8, 16, 16);
                }

                // ── % text (vertically centered in arc half-circle) ───
                // Arc center is at cy; the arc spans from cy-r (top) to cy (bottom)
                // So the visual center of the arc area is cy - r/2
                int textCY = cy - r / 2;

                string pctText = goal > 0 ? $"{(int)(pct * 100)}%" : "—";
                using var pctFont = new Font("Segoe UI", 26f, FontStyle.Bold);
                using var pctBrush = new SolidBrush(Color.FromArgb(25, 25, 25));
                SizeF pctSz = g.MeasureString(pctText, pctFont);
                g.DrawString(pctText, pctFont, pctBrush,
                    cx - pctSz.Width / 2,
                    textCY - pctSz.Height / 2);

                // "of goal" — directly below %, fixed 4px gap
                using var subFont = new Font("Segoe UI", 8.5f, FontStyle.Regular);
                using var subBrush = new SolidBrush(Color.FromArgb(150, 150, 162));
                SizeF subSz = g.MeasureString("of goal", subFont);
                float subY = textCY - pctSz.Height / 2 + pctSz.Height + 4f;
                g.DrawString("of goal", subFont, subBrush,
                    cx - subSz.Width / 2, subY);

                // Status message — fixed 6px below "of goal"
                string msg = goal <= 0 ? "No goal set"
                               : achieved ? "Goal achieved!"
                                            : $"{goal - todayTotal:F0} kcal left";
                Color msgCol = goal <= 0 ? Color.FromArgb(160, 160, 170)
                               : achieved ? Color.FromArgb(40, 167, 69)
                                            : Color.FromArgb(210, 120, 0);

                using var msgFont = new Font("Segoe UI", 9f, FontStyle.Regular);
                using var msgBrush = new SolidBrush(msgCol);
                SizeF msgSz = g.MeasureString(msg, msgFont);
                g.DrawString(msg, msgFont, msgBrush,
                    cx - msgSz.Width / 2,
                    subY + subSz.Height + 6f);

                // Vertical divider
                int divX = cx + r + 26;
                using var divPen = new Pen(Color.FromArgb(228, 228, 234), 1);
                g.DrawLine(divPen, divX, 14, divX, H - 14);
            };

            // ── Four mini stat cards ───────────────────────────────────
            int dividerX = 110 + 80 + 26;
            int rightX = dividerX + 20;
            int availW = W - rightX - 16;
            int miniW = (availW - 30) / 4;
            int miniH = 116;
            int miniY = (H - miniH) / 2;

            string[] labels = { "Today Burned", "Calorie Goal", "Today Sessions", "All-Time Burned" };
            string[] values =
            {
                $"{todayTotal:F0} kcal",
                goal > 0 ? $"{goal:F0} kcal" : "Not Set",
                todayCount.ToString(),
                $"{allTime:F0} kcal"
            };
            Color[] colors =
            {
                Color.FromArgb(255, 140,   0),
                Color.FromArgb( 13, 110, 253),
                Color.FromArgb(111,  66, 193),
                Color.FromArgb( 32, 178, 170)
            };

            for (int i = 0; i < 4; i++)
            {
                string lbl = labels[i];
                string val = values[i];
                Color col = colors[i];
                int mx = rightX + i * (miniW + 10);

                var mini = new Panel
                {
                    Left = mx,
                    Top = miniY,
                    Width = miniW,
                    Height = miniH,
                    BackColor = Color.FromArgb(247, 248, 250)
                };

                mini.Paint += (ms, me) =>
                {
                    var g = me.Graphics;
                    g.SmoothingMode = System.Drawing.Drawing2D.SmoothingMode.AntiAlias;

                    using var ab = new SolidBrush(col);
                    g.FillRectangle(ab, 0, 0, mini.Width, 3);

                    using var pb = new Pen(Color.FromArgb(222, 222, 228), 1);
                    g.DrawRectangle(pb, 0, 0, mini.Width - 1, mini.Height - 1);

                    using var lf = new Font("Segoe UI", 8.5f, FontStyle.Regular);
                    using var lb = new SolidBrush(Color.FromArgb(118, 118, 130));
                    g.DrawString(lbl, lf, lb, 12, 16);

                    using var vf = new Font("Segoe UI", 15f, FontStyle.Bold);
                    using var vb = new SolidBrush(Color.FromArgb(25, 25, 25));
                    g.DrawString(val, vf, vb, 12, 42);

                    using var db2 = new SolidBrush(Color.FromArgb(38, col.R, col.G, col.B));
                    using var dp = new Pen(col, 1.5f);
                    g.FillEllipse(db2, mini.Width - 22, 13, 10, 10);
                    g.DrawEllipse(dp, mini.Width - 22, 13, 10, 10);
                };

                card.Controls.Add(mini);
            }

            pnlMain.Controls.Add(card);
        }

        // ─────────────────────────────────────────────────────────────
        //  Weekly Bar Chart
        // ─────────────────────────────────────────────────────────────
        void AddWeeklyChart(int top, List<ActivityEntry> allActivities)
        {
            int W = pnlMain.Width - SideMargin * 2;
            int H = 140;

            pnlMain.Controls.Add(SectionLabel("This Week", SideMargin, top));

            var chart = WhiteCard(SideMargin, top + 30, W, H);

            // Build Mon–Sun
            var startOfWeek = DateTime.Today.DayOfWeek == DayOfWeek.Sunday
                ? DateTime.Today.AddDays(-6)
                : DateTime.Today.AddDays(-(int)DateTime.Today.DayOfWeek + 1);

            string[] dayLabels = { "Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun" };
            double[] dayTotals = new double[7];
            for (int d = 0; d < 7; d++)
                dayTotals[d] = allActivities
                    .Where(a => a.Date.Date == startOfWeek.AddDays(d))
                    .Sum(a => a.CaloriesBurned);

            double maxVal = dayTotals.Max();
            int todayIdx = ((int)DateTime.Today.DayOfWeek + 6) % 7;

            chart.Paint += (s, e) =>
            {
                var g = e.Graphics;
                g.SmoothingMode = System.Drawing.Drawing2D.SmoothingMode.AntiAlias;

                using var border = new Pen(Color.FromArgb(218, 218, 224), 1);
                g.DrawRectangle(border, 0, 0, chart.Width - 1, chart.Height - 1);

                int padX = 24, padY = 18, labelH = 24;
                int areaH = H - padY * 2 - labelH;
                int totalW = chart.Width - padX * 2;
                int barW = totalW / 7 - 10;
                int barGap = (totalW - barW * 7) / 7;

                for (int d = 0; d < 7; d++)
                {
                    int bx = padX + d * (barW + barGap) + barGap / 2;
                    bool isToday = d == todayIdx;
                    Color barCol = isToday
                        ? Color.FromArgb(13, 110, 253)
                        : Color.FromArgb(198, 212, 240);
                    int bh = maxVal > 0
                        ? Math.Max(4, (int)(areaH * (dayTotals[d] / maxVal)))
                        : 4;
                    int by = padY + areaH - bh;

                    using var bb = new SolidBrush(barCol);
                    g.FillRectangle(bb, bx, by, barW, bh);

                    using var df = new Font("Segoe UI", 8f, FontStyle.Regular);
                    Color dc = isToday
                        ? Color.FromArgb(13, 110, 253)
                        : Color.FromArgb(128, 128, 140);
                    using var db = new SolidBrush(dc);
                    SizeF ls = g.MeasureString(dayLabels[d], df);
                    g.DrawString(dayLabels[d], df, db,
                        bx + (barW - ls.Width) / 2, padY + areaH + 4);

                    if (dayTotals[d] > 0)
                    {
                        string cv = dayTotals[d] >= 1000
                            ? $"{dayTotals[d] / 1000:F1}k" : $"{dayTotals[d]:F0}";
                        using var cf = new Font("Segoe UI", 7.5f, FontStyle.Regular);
                        using var cb = new SolidBrush(Color.FromArgb(100, 100, 112));
                        SizeF cs = g.MeasureString(cv, cf);
                        g.DrawString(cv, cf, cb,
                            bx + (barW - cs.Width) / 2, by - cs.Height - 1);
                    }
                }
            };

            pnlMain.Controls.Add(chart);
        }

        // ─────────────────────────────────────────────────────────────
        //  Quick Stats Row
        // ─────────────────────────────────────────────────────────────
        void AddQuickStats(int top, List<ActivityEntry> all)
        {
            int W = pnlMain.Width - SideMargin * 2;

            pnlMain.Controls.Add(SectionLabel("Quick Stats", SideMargin, top));

            int totalSessions = all.Count;
            double avgCal = totalSessions > 0 ? all.Average(a => a.CaloriesBurned) : 0;
            string bestDay = "—";

            if (all.Count > 0)
            {
                var best = all
                    .GroupBy(a => a.Date.Date)
                    .Select(g => new { Day = g.Key, Total = g.Sum(a => a.CaloriesBurned) })
                    .OrderByDescending(x => x.Total).First();
                bestDay = $"{best.Day:ddd dd MMM}  ({best.Total:F0} kcal)";
            }

            string[] slabels = { "Total Sessions", "Avg Cal / Session", "Best Day" };
            string[] svalues = { totalSessions.ToString(), $"{avgCal:F0} kcal", bestDay };
            Color[] scolors =
            {
                Color.FromArgb(111,  66, 193),
                Color.FromArgb( 32, 178, 170),
                Color.FromArgb(255, 140,   0)
            };

            int cW = (W - 20) / 3;
            int cH = 76;

            for (int i = 0; i < 3; i++)
            {
                string lbl = slabels[i], val = svalues[i];
                Color col = scolors[i];
                var c = WhiteCard(SideMargin + i * (cW + 10), top + 30, cW, cH);

                c.Paint += (s, e) =>
                {
                    var g = e.Graphics;
                    g.SmoothingMode = System.Drawing.Drawing2D.SmoothingMode.AntiAlias;

                    using var sb = new SolidBrush(col);
                    g.FillRectangle(sb, 0, 0, 4, cH);

                    using var pb = new Pen(Color.FromArgb(222, 222, 228), 1);
                    g.DrawRectangle(pb, 0, 0, c.Width - 1, cH - 1);

                    using var lf = new Font("Segoe UI", 8.5f, FontStyle.Regular);
                    using var lb = new SolidBrush(Color.FromArgb(118, 118, 130));
                    g.DrawString(lbl, lf, lb, 16, 10);

                    using var vf = new Font("Segoe UI", 14f, FontStyle.Bold);
                    using var vb = new SolidBrush(Color.FromArgb(25, 25, 25));
                    g.DrawString(val, vf, vb, 16, 34);
                };

                pnlMain.Controls.Add(c);
            }
        }

        // ─────────────────────────────────────────────────────────────
        //  Personal Records
        // ─────────────────────────────────────────────────────────────
        void AddPersonalRecords(int top, List<ActivityEntry> all)
        {
            int W = pnlMain.Width - SideMargin * 2;

            pnlMain.Controls.Add(SectionLabel("Personal Records", SideMargin, top));

            // Best single session per activity type
            var records = all
                .GroupBy(a => a.ActivityName)
                .Select(g => new
                {
                    Name = g.Key,
                    Best = g.Max(a => a.CaloriesBurned),
                    Count = g.Count()
                })
                .OrderByDescending(r => r.Best)
                .ToList();

            if (records.Count == 0)
            {
                pnlMain.Controls.Add(new Label
                {
                    Text = "No activities recorded yet.",
                    Left = SideMargin,
                    Top = top + 32,
                    AutoSize = true,
                    Font = new Font("Segoe UI", 9.5f, FontStyle.Regular),
                    ForeColor = Color.FromArgb(150, 150, 162),
                    BackColor = Color.Transparent
                });
                return;
            }

            Color[] medalColors =
            {
                Color.FromArgb(212, 175,  55),  // gold
                Color.FromArgb(160, 160, 160),  // silver
                Color.FromArgb(176, 115,  65),  // bronze
            };

            int cardW = Math.Min(200, (W - (records.Count - 1) * 12) / records.Count);
            int cardH = 90;
            int cardsY = top + 30;

            for (int i = 0; i < records.Count; i++)
            {
                var rec = records[i];
                Color col = i < 3 ? medalColors[i] : Color.FromArgb(13, 110, 253);
                int cx = SideMargin + i * (cardW + 12);
                var card = WhiteCard(cx, cardsY, cardW, cardH);

                card.Paint += (s, e) =>
                {
                    var g = e.Graphics;
                    g.SmoothingMode = System.Drawing.Drawing2D.SmoothingMode.AntiAlias;

                    using var pb = new Pen(Color.FromArgb(222, 222, 228), 1);
                    g.DrawRectangle(pb, 0, 0, card.Width - 1, cardH - 1);

                    // Top medal stripe
                    using var sb = new SolidBrush(col);
                    g.FillRectangle(sb, 0, 0, card.Width, 3);

                    // Rank badge
                    string rank = i == 0 ? "🥇" : i == 1 ? "🥈" : i == 2 ? "🥉" : $"#{i + 1}";
                    using var rf = new Font("Segoe UI", 9f, FontStyle.Regular);
                    using var rb = new SolidBrush(Color.FromArgb(100, 100, 112));
                    g.DrawString(rank, rf, rb, 10, 10);

                    // Activity name
                    using var nf = new Font("Segoe UI", 9.5f, FontStyle.Bold);
                    using var nb = new SolidBrush(Color.FromArgb(35, 35, 45));
                    g.DrawString(rec.Name, nf, nb, 10, 30);

                    // Best calories
                    using var bf = new Font("Segoe UI", 13f, FontStyle.Bold);
                    using var bb = new SolidBrush(col);
                    g.DrawString($"{rec.Best:F0} kcal", bf, bb, 10, 52);

                    // Session count
                    using var cf = new Font("Segoe UI", 7.5f, FontStyle.Regular);
                    using var cb = new SolidBrush(Color.FromArgb(140, 140, 152));
                    string sc = rec.Count == 1 ? "1 session" : $"{rec.Count} sessions";
                    SizeF ss = g.MeasureString(sc, cf);
                    g.DrawString(sc, cf, cb,
                        card.Width - ss.Width - 10, cardH - ss.Height - 8);
                };

                pnlMain.Controls.Add(card);
            }
        }

        // ─────────────────────────────────────────────────────────────
        //  Full Activity Log
        // ─────────────────────────────────────────────────────────────
        void AddActivityLog(int top, List<ActivityEntry> all)
        {
            int W = pnlMain.Width - SideMargin * 2;
            int rowHeight = 36;

            pnlMain.Controls.Add(SectionLabel("Activity Log", SideMargin, top));

            if (all.Count == 0)
            {
                pnlMain.Controls.Add(new Label
                {
                    Text = "No activities recorded yet.",
                    Left = SideMargin,
                    Top = top + 32,
                    AutoSize = true,
                    Font = new Font("Segoe UI", 9.5f, FontStyle.Regular),
                    ForeColor = Color.FromArgb(150, 150, 162),
                    BackColor = Color.Transparent
                });
                return;
            }

            int tableTop = top + 30;

            // Header
            var header = WhiteCard(SideMargin, tableTop, W, rowHeight);
            header.BackColor = Color.FromArgb(238, 239, 242);

            header.Paint += (s, e) =>
            {
                using var pen = new Pen(Color.FromArgb(218, 218, 224));
                e.Graphics.DrawRectangle(pen, 0, 0, header.Width - 1, header.Height - 1);
            };

            AddLogCell(header, "Date", 0, W, rowHeight, true);
            AddLogCell(header, "Activity", W / 5, W, rowHeight, true);
            AddLogCell(header, "Details", W * 2 / 5, W, rowHeight, true);
            AddLogCell(header, "Calories", W * 4 / 5, W, rowHeight, true);
            pnlMain.Controls.Add(header);

            // Rows — newest first
            var sorted = all.OrderByDescending(a => a.Date).ToList();

            for (int i = 0; i < sorted.Count; i++)
            {
                var a = sorted[i];
                int rowTop = tableTop + rowHeight + i * rowHeight;
                Color rowBg = i % 2 == 0 ? Color.White : Color.FromArgb(249, 249, 252);

                var row = new Panel
                {
                    Left = SideMargin,
                    Top = rowTop,
                    Width = W,
                    Height = rowHeight,
                    BackColor = rowBg
                };

                row.Paint += (s, e) =>
                {
                    using var pen = new Pen(Color.FromArgb(232, 232, 236));
                    e.Graphics.DrawLine(pen, 0, row.Height - 1, row.Width, row.Height - 1);
                };

                string details =
                    $"{a.Metric1Label}: {a.Metric1}  " +
                    $"{a.Metric2Label}: {a.Metric2}  " +
                    $"{a.Metric3Label}: {a.Metric3}";

                AddLogCell(row, a.Date.ToString("ddd dd MMM, HH:mm"), 0, W, rowHeight, false);
                AddLogCell(row, a.ActivityName, W / 5, W, rowHeight, false);
                AddLogCell(row, details, W * 2 / 5, W, rowHeight, false);

                // Calorie cell — colored
                row.Controls.Add(new Label
                {
                    Text = $"{a.CaloriesBurned:F1} kcal",
                    Left = W * 4 / 5 + 12,
                    Top = 0,
                    Width = W / 5 - 12,
                    Height = rowHeight,
                    Font = new Font("Segoe UI", 9f, FontStyle.Bold),
                    ForeColor = Color.FromArgb(255, 140, 0),
                    TextAlign = ContentAlignment.MiddleLeft,
                    BackColor = Color.Transparent
                });

                pnlMain.Controls.Add(row);
            }
        }

        void AddLogCell(Panel parent, string text, int x,
                        int W, int rowH, bool isHeader)
        {
            parent.Controls.Add(new Label
            {
                Text = text,
                Left = x + 12,
                Top = 0,
                Width = W / 5 - 12,
                Height = rowH,
                Font = isHeader
                    ? new Font("Segoe UI", 9f, FontStyle.Bold)
                    : new Font("Segoe UI", 9f, FontStyle.Regular),
                ForeColor = isHeader
                    ? Color.FromArgb(75, 75, 88)
                    : Color.FromArgb(38, 38, 50),
                TextAlign = ContentAlignment.MiddleLeft,
                BackColor = Color.Transparent
            });
        }

        // ─────────────────────────────────────────────────────────────
        //  Calorie Breakdown Table
        // ─────────────────────────────────────────────────────────────
        void AddActivityBreakdown(int top, List<ActivityEntry> activities)
        {
            int W = pnlMain.Width - SideMargin * 2;

            pnlMain.Controls.Add(SectionLabel("Today's Calorie Breakdown", SideMargin, top));

            var grouped = activities
                .GroupBy(a => a.ActivityName)
                .Select(g => new
                {
                    Name = g.Key,
                    Count = g.Count(),
                    Calories = g.Sum(a => a.CaloriesBurned)
                })
                .OrderByDescending(g => g.Calories)
                .ToList();

            if (grouped.Count == 0)
            {
                pnlMain.Controls.Add(new Label
                {
                    Text = "No activities recorded today.",
                    Left = SideMargin,
                    Top = top + 32,
                    AutoSize = true,
                    Font = new Font("Segoe UI", 9.5f, FontStyle.Regular),
                    ForeColor = Color.FromArgb(150, 150, 162),
                    BackColor = Color.Transparent
                });
                return;
            }

            int tableTop = top + 30;
            int rowHeight = 38;

            var header = WhiteCard(SideMargin, tableTop, W, rowHeight);
            header.BackColor = Color.FromArgb(238, 239, 242);

            header.Paint += (s, e) =>
            {
                using var pen = new Pen(Color.FromArgb(218, 218, 224));
                e.Graphics.DrawRectangle(pen, 0, 0, header.Width - 1, header.Height - 1);
            };

            AddTableCell(header, "Activity", 0, W, rowHeight, true);
            AddTableCell(header, "Sessions", W / 2, W, rowHeight, true);
            AddTableCell(header, "Calories Burned", W * 3 / 4, W, rowHeight, true);
            pnlMain.Controls.Add(header);

            double maxCal = grouped.Max(g => g.Calories);

            for (int i = 0; i < grouped.Count; i++)
            {
                var item = grouped[i];
                int rowTop = tableTop + rowHeight + i * rowHeight;
                Color rowBg = i % 2 == 0 ? Color.White : Color.FromArgb(249, 249, 252);

                var row = new Panel
                {
                    Left = SideMargin,
                    Top = rowTop,
                    Width = W,
                    Height = rowHeight,
                    BackColor = rowBg
                };

                row.Paint += (s, e) =>
                {
                    using var pen = new Pen(Color.FromArgb(232, 232, 236));
                    e.Graphics.DrawLine(pen, 0, row.Height - 1, row.Width, row.Height - 1);
                };

                AddTableCell(row, item.Name, 0, W, rowHeight, false);
                AddTableCell(row,
                    item.Count == 1 ? "1 session" : $"{item.Count} sessions",
                    W / 2, W, rowHeight, false);

                int calX = W * 3 / 4;
                int barMaxW = W / 4 - 90;
                int barW = maxCal > 0 ? (int)(barMaxW * (item.Calories / maxCal)) : 0;

                row.Controls.Add(new Label
                {
                    Text = $"{item.Calories:F1} kcal",
                    Left = calX,
                    Top = 0,
                    Width = 85,
                    Height = rowHeight,
                    Font = new Font("Segoe UI", 9f, FontStyle.Bold),
                    ForeColor = Color.FromArgb(255, 140, 0),
                    TextAlign = ContentAlignment.MiddleLeft,
                    BackColor = Color.Transparent
                });

                if (barW > 0)
                    row.Controls.Add(new Panel
                    {
                        Left = calX + 88,
                        Top = (rowHeight - 8) / 2,
                        Width = barW,
                        Height = 8,
                        BackColor = Color.FromArgb(255, 208, 112)
                    });

                pnlMain.Controls.Add(row);
            }
        }

        void AddTableCell(Panel parent, string text, int x,
                          int W, int rowH, bool isHeader)
        {
            parent.Controls.Add(new Label
            {
                Text = text,
                Left = x + 12,
                Top = 0,
                Width = W / 4 - 12,
                Height = rowH,
                Font = isHeader
                    ? new Font("Segoe UI", 9f, FontStyle.Bold)
                    : new Font("Segoe UI", 9.5f, FontStyle.Regular),
                ForeColor = isHeader
                    ? Color.FromArgb(75, 75, 88)
                    : Color.FromArgb(38, 38, 50),
                TextAlign = ContentAlignment.MiddleLeft,
                BackColor = Color.Transparent
            });
        }

        // ─────────────────────────────────────────────────────────────
        //  Dashboard Rendering
        // ─────────────────────────────────────────────────────────────
        void ShowDashboard()
        {
            var u = UserStore.CurrentUser;
            if (u == null) { MessageBox.Show("No user session found."); return; }

            lblWelcome.Text = $"Hello, {u.Username} 👋";
            pnlMain.Controls.Clear();

            // ── Page heading ──────────────────────────────────────────
            pnlMain.Controls.Add(new Label
            {
                Text = "Dashboard",
                ForeColor = Color.FromArgb(22, 22, 28),
                Font = new Font("Segoe UI", 15f, FontStyle.Bold),
                Left = SideMargin,
                Top = 18,
                AutoSize = true,
                BackColor = Color.Transparent
            });

            pnlMain.Controls.Add(new Label
            {
                Text = $"📅  {DateTime.Today:dddd, dd MMMM yyyy}",
                ForeColor = Color.FromArgb(128, 128, 142),
                Font = new Font("Segoe UI", 9f, FontStyle.Regular),
                Left = SideMargin,
                Top = 50,   // fixed gap — won't overlap title
                AutoSize = true,
                BackColor = Color.Transparent
            });

            // ── Compute data ──────────────────────────────────────────
            double todayTotal = u.GetTodayCalories();
            int todayCount = u.GetTodayActivityCount();
            double goal = u.CalorieGoal;
            double allTime = u.GetTotalCalories();

            // ── Layout: stack each section with explicit y positions ──
            int y = 80;

            AddGoalProgressSection(y, todayTotal, goal, todayCount, allTime);
            y += 170 + SectionGap;

            AddWeeklyChart(y, u.Activities);
            y += 30 + 140 + SectionGap;

            AddQuickStats(y, u.Activities);
            y += 30 + 76 + SectionGap;

            // Divider
            pnlMain.Controls.Add(new Panel
            {
                Left = SideMargin,
                Top = y,
                Width = pnlMain.Width - SideMargin * 2,
                Height = 1,
                BackColor = Color.FromArgb(218, 218, 224)
            });
            y += 14;

            AddActivityBreakdown(y, u.Activities
                .Where(a => a.Date.Date == DateTime.Today).ToList());

            // Count today's rows for offset
            int todayRows = u.Activities.Count(a => a.Date.Date == DateTime.Today);
            y += 30 + 38 + todayRows * 38 + SectionGap;

            // Divider
            pnlMain.Controls.Add(new Panel
            {
                Left = SideMargin,
                Top = y,
                Width = pnlMain.Width - SideMargin * 2,
                Height = 1,
                BackColor = Color.FromArgb(218, 218, 224)
            });
            y += 14;

            AddPersonalRecords(y, u.Activities);

            int recCount = u.Activities.Select(a => a.ActivityName).Distinct().Count();
            y += 30 + 90 + SectionGap;

            // Divider
            pnlMain.Controls.Add(new Panel
            {
                Left = SideMargin,
                Top = y,
                Width = pnlMain.Width - SideMargin * 2,
                Height = 1,
                BackColor = Color.FromArgb(218, 218, 224)
            });
            y += 14;

            AddActivityLog(y, u.Activities);
        }

        // ─────────────────────────────────────────────────────────────
        //  Logout
        // ─────────────────────────────────────────────────────────────
        void OnLogout(object sender, EventArgs e)
        {
            var confirm = MessageBox.Show(
                "Are you sure you want to logout?",
                "Logout",
                MessageBoxButtons.YesNo,
                MessageBoxIcon.Question);

            if (confirm != DialogResult.Yes) return;

            UserStore.Logout();
            new LoginForm().Show();
            this.Close();
        }
