    /// <summary>
    /// Represents a registered user account.
    /// Stores credentials, calorie goal, lockout state,
    /// and the list of all recorded fitness activities.
    /// </summary>
    class UserAccount
    {
        /// <summary>Unique login identifier for the user.</summary>
        public string Username { get; set; }

        /// <summary>SHA-256 hashed password (never stored as plain text).</summary>
        public string PasswordHash { get; set; }

        /// <summary>Target calories the user wants to burn per day.</summary>
        public double CalorieGoal { get; set; }

        /// <summary>Tracks how many consecutive wrong passwords were entered.</summary>
        public int FailedAttempts { get; set; }

        /// <summary>
        /// When true, the account is locked and login is blocked.
        /// Triggered after 3 consecutive failed login attempts.
        /// </summary>
        public bool IsLocked { get; set; }

        /// <summary>All fitness activities ever recorded by this user.</summary>
        public List<ActivityEntry> Activities { get; set; } = new List<ActivityEntry>();
    }

    /// <summary>
    /// Represents a single fitness activity session recorded by the user.
    /// Stores the activity type, the three metric values entered,
    /// the calculated calories burned, and the date of the session.
    /// </summary>
    class ActivityEntry
    {
        /// <summary>Name of the activity (e.g. "Walking", "Running").</summary>
        public string ActivityName { get; set; }

        /// <summary>First metric value (meaning depends on activity type).</summary>
        public double Metric1 { get; set; }

        /// <summary>Second metric value (meaning depends on activity type).</summary>
        public double Metric2 { get; set; }

        /// <summary>Third metric value (meaning depends on activity type).</summary>
        public double Metric3 { get; set; }

        /// <summary>Calories burned, calculated from the three metrics using the activity formula.</summary>
        public double CaloriesBurned { get; set; }

        /// <summary>The date on which this activity was recorded.</summary>
        public DateTime Date { get; set; }
    }

    /// <summary>
    /// Provides password hashing using SHA-256.
    /// Passwords are NEVER stored or compared as plain text —
    /// only their hash values are used.
    /// </summary>
    static class Security
    {
        /// <summary>
        /// Converts a plain-text password into a SHA-256 hex digest.
        /// The same password always produces the same hash,
        /// but the original password cannot be recovered from the hash.
        /// </summary>
        /// <param name="password">The plain-text password to hash.</param>
        /// <returns>64-character lowercase hex string.</returns>
        public static string HashPassword(string password)
        {
            using (var sha = SHA256.Create())
            {
                // Convert password string to bytes, compute SHA-256, convert to hex string
                byte[] bytes = sha.ComputeHash(Encoding.UTF8.GetBytes(password));
                var sb = new StringBuilder();
                foreach (byte b in bytes)
                    sb.Append(b.ToString("x2")); // "x2" = two-digit lowercase hex
                return sb.ToString();
            }
        }
    }

    // ──────────────────────────────────────────────────────────
    //  USER STORE  (JSON persistence — Facade pattern)
    //  All file I/O is encapsulated here. No other class
    //  needs to know about the JSON file or its path.
    // ──────────────────────────────────────────────────────────

    /// <summary>
    /// Static service class that manages the in-memory user database
    /// and handles JSON serialisation / deserialisation to disk.
    /// Implements the Facade pattern: simple Load/Save interface
    /// over file I/O details.
    /// </summary>
    static class UserStore
    {
        // Path to the JSON data file (relative to the executable location)
        private static readonly string FilePath = "users.json";

        /// <summary>In-memory dictionary of all user accounts, keyed by username.</summary>
        public static Dictionary<string, UserAccount> Users =
            new Dictionary<string, UserAccount>();

        /// <summary>The account of the currently logged-in user. Null when logged out.</summary>
        public static UserAccount CurrentUser = null;

        /// <summary>
        /// Loads user accounts from the JSON file on disk into memory.
        /// Called once at application startup.
        /// If the file does not exist, starts with an empty dictionary.
        /// </summary>
        public static void Load()
        {
            if (!File.Exists(FilePath))
            {
                Users = new Dictionary<string, UserAccount>();
                return;
            }

            try
            {
                string json = File.ReadAllText(FilePath);
                var opts     = new JsonSerializerOptions { PropertyNameCaseInsensitive = true };
                Users        = JsonSerializer.Deserialize<Dictionary<string, UserAccount>>(json, opts)
                               ?? new Dictionary<string, UserAccount>();
            }
            catch
            {
                // If the file is corrupt or unreadable, start fresh
                Users = new Dictionary<string, UserAccount>();
            }
        }

        /// <summary>
        /// Saves the current in-memory user dictionary to the JSON file on disk.
        /// Called after every data-changing operation (register, login, save activity, etc.).
        /// </summary>
        public static void Save()
        {
            try
            {
                var opts = new JsonSerializerOptions { WriteIndented = true };
                File.WriteAllText(FilePath, JsonSerializer.Serialize(Users, opts));
            }
            catch (Exception ex)
            {
                // Warn the user if saving fails — data may be lost on exit
                MessageBox.Show($"Warning: Could not save data.\n{ex.Message}",
                    "Save Error", MessageBoxButtons.OK, MessageBoxIcon.Warning);
            }
        }
    }

        /// <summary>
        /// Handles the LOGIN button click.
        /// Validates credentials, tracks failed attempts,
        /// and locks the account after 3 failures.
        /// On success, opens the Dashboard.
        /// </summary>
        void BtnLogin_Click(object sender, EventArgs e)
        {
            lblError.Text = "";
            string user = tbUser.Text.Trim();
            string pass = tbPass.Text;

            // Step 1: Check if the username exists in the database
            if (!UserStore.Users.ContainsKey(user))
            {
                lblError.Text = "⚠ Username not found.";
                return;
            }

            var acc = UserStore.Users[user];

            // Step 2: Check if the account is already locked
            if (acc.IsLocked)
            {
                lblError.Text = "🔒 Account locked. Please contact support.";
                return;
            }

            // Step 3: Compare the hashed password
            string enteredHash = Security.HashPassword(pass);
            if (acc.PasswordHash != enteredHash)
            {
                acc.FailedAttempts++;

                // Lock the account after 3 consecutive failures
                if (acc.FailedAttempts >= 3)
                {
                    acc.IsLocked = true;
                    lblError.Text = "🔒 Locked after 3 failed attempts.";
                }
                else
                {
                    lblError.Text = $"✗ Wrong password. Attempt {acc.FailedAttempts}/3.";
                }

                UserStore.Save(); // Persist the updated failed-attempt count
                return;
            }

            // Step 4: Login successful — reset failed attempts and open Dashboard
            acc.FailedAttempts  = 0;
            UserStore.CurrentUser = acc;
            UserStore.Save();

            new DashboardForm().Show();
            Hide(); // Hide login form (don't close — user may log out and return)
        }
    }

            btn.Left = 30; btn.Top = y; btn.Width = 320;
            card.Controls.Add(btn);
            btn.Click += Btn_Click;
        }

        /// <summary>
        /// Validates all registration fields and creates a new account if all rules pass.
        ///
        /// Validation rules (from assignment brief):
        ///   - Username: letters and numbers only, minimum 3 characters, must be unique
        ///   - Password: minimum 12 characters, at least 1 uppercase, at least 1 lowercase
        ///   - Confirm password must match password
        /// </summary>
        void Btn_Click(object sender, EventArgs e)
        {
            lblError.Text   = "";
            lblSuccess.Text = "";

            string user = tbUser.Text.Trim();
            string pass = tbPass.Text;
            string conf = tbConfirm.Text;

            // FIX: Check for empty username FIRST (before regex) for a clearer error message
            if (string.IsNullOrEmpty(user))
            {
                lblError.Text = "✗ Username cannot be empty.";
                return;
            }

            // Username must contain only letters and numbers (no spaces or special chars)
            if (!Regex.IsMatch(user, @"^[a-zA-Z0-9]+$"))
            {
                lblError.Text = "✗ Letters and numbers only.";
                return;
            }

            // Username minimum length
            if (user.Length < 3)
            {
                lblError.Text = "✗ Username min 3 characters.";
                return;
            }

            // Username must not already exist
            if (UserStore.Users.ContainsKey(user))
            {
                lblError.Text = "✗ Username already exists.";
                return;
            }

            // Password minimum length (12 characters as per brief)
            if (pass.Length < 12)
            {
                lblError.Text = "✗ Password min 12 characters.";
                return;
            }

            // Password must contain at least one uppercase letter
            if (!Regex.IsMatch(pass, @"[A-Z]"))
            {
                lblError.Text = "✗ Need at least 1 uppercase letter.";
                return;
            }

            // Password must contain at least one lowercase letter
            if (!Regex.IsMatch(pass, @"[a-z]"))
            {
                lblError.Text = "✗ Need at least 1 lowercase letter.";
                return;
            }

            // Confirm password must match
            if (pass != conf)
            {
                lblError.Text = "✗ Passwords do not match.";
                return;
            }

            // All validation passed — create account with HASHED password
            UserStore.Users[user] = new UserAccount
            {
                Username     = user,
                PasswordHash = Security.HashPassword(pass)  // FIX: Store hash, not plain text
            };
            UserStore.Save();

            lblSuccess.Text             = "✔ Account created! You can now login.";
            tbUser.Text = tbPass.Text   = tbConfirm.Text = "";
        }
    }

        /// <summary>
        /// Refreshes the main panel with today's stats:
        /// calories burned, goal, remaining, activity count, and activity list.
        /// </summary>
        public void ShowDashboard()
        {
            var u = UserStore.CurrentUser;
            lblWelcome.Text = $"Hello, {u.Username}";
            pnlMain.Controls.Clear();

            // Header
            pnlMain.Controls.Add(new Label
            {
                Text = "Dashboard Overview", ForeColor = Theme.TextPrimary,
                Font = Theme.FontTitle, Left = 30, Top = 30,
                AutoSize = true, BackColor = Color.Transparent
            });
            pnlMain.Controls.Add(Theme.StyledLabel(
                $"📅 {DateTime.Today:dddd, dd MMMM yyyy}", 30, 72, Theme.FontBody, Theme.TextSec));

            // ── Calculate today's statistics ──
            // Sum calories from activities recorded today only
            double todayTotal = u.Activities
                .Where(a => a.Date.Date == DateTime.Today)
                .Sum(a => a.CaloriesBurned);

            int todayCount = u.Activities
                .Count(a => a.Date.Date == DateTime.Today);

            double goal      = u.CalorieGoal;
            double remaining = Math.Max(0, goal - todayTotal);
            bool achieved    = goal > 0 && todayTotal >= goal;

            // ── Stat cards row 1 ──

    /// <summary>
    /// Displays metric input fields for each selected activity
    /// and calculates / saves calories burned.
    ///
    /// The ActivityDefs dictionary maps each activity to:
    ///   - Labels for its three metrics
    ///   - A lambda function that computes calories from those metrics
    ///
    /// This data-driven design (Strategy pattern) means adding a new
    /// activity only requires one new dictionary entry — no new forms or
    /// if-else chains needed.
    /// </summary>
    class MultiCalcForm : Form
    {
        List<string> selectedActivities;

        // Stores the three metric TextBoxes for each activity in order
        List<(TextBox m1, TextBox m2, TextBox m3)> metricBoxes =
            new List<(TextBox, TextBox, TextBox)>();

        Panel pnlScroll;
        Label lblResult;

        /// <summary>Callback invoked after activities are saved successfully.</summary>
        public Action OnSaved;

        // ── Activity definitions dictionary ──
        // Each entry maps an activity name to:
        //   (metric1 label, metric2 label, metric3 label, calorie formula)
        //
        // FORMULA SOURCES & FIXES:
        //   Walking:   Calories = (steps × 0.04) + (distance_km × 30) + (time_min × 3.5)
        //              All 3 metrics used. Step factor ~0.04 kcal/step (avg person).
        //              FIX: Previously only used steps.
        //
        //   Swimming:  Calories = (laps × 8) + (time_min × 0.5)
        //              Lap factor ~8 kcal/lap (25m pool, moderate pace).
        //
        //   Running:   Calories = (distance_km × 65) + (speed_kmh × 0.5)
        //              FIX: Previous formula gave ~22 kcal for 5 km (wrong).
        //              New formula gives ~325 kcal for 5 km (realistic).
        //
        //   Cycling:   Calories = (distance_km × 8) + (speed_kmh × 3)
        //              FIX: Previous formula gave ~10 kcal for 10 km at 20 km/h (wrong).
        //              New formula gives ~140 kcal (realistic for moderate cycling).
        //
        //   Pushups:   Calories = reps × sets × 0.35
        //              ~0.35 kcal per rep (standard estimate).
        //
        //   Jump Rope: Calories = (jumps × 0.012) + (time_min × 0.8) + (heart_rate × 0.05)
        //              FIX: Previously ignored heart rate (m3). Now uses all 3 metrics.
        static readonly Dictionary<string,
            (string m1, string m2, string m3, Func<double, double, double, double> calc)>
            ActivityDefs = new Dictionary<string,
            (string, string, string, Func<double, double, double, double>)>
        {
            ["Walking"]   = ("Steps",          "Distance (km)",  "Time (min)",
                             (m1, m2, m3) => (m1 * 0.04) + (m2 * 30) + (m3 * 3.5)),

            ["Swimming"]  = ("Laps",           "Time (min)",     "Heart Rate",
                             (m1, m2, m3) => m1 * 8.0 + m2 * 0.5),

            ["Running"]   = ("Distance (km)",  "Time (min)",     "Speed (km/h)",
                             (m1, m2, m3) => (m1 * 65) + (m3 * 0.5)),

            ["Cycling"]   = ("Distance (km)",  "Time (min)",     "Speed (km/h)",
                             (m1, m2, m3) => (m1 * 8) + (m3 * 3)),

            ["Pushups"]   = ("Reps",           "Sets",           "Time (min)",
                             (m1, m2, m3) => m1 * m2 * 0.35),

            ["Jump Rope"] = ("Jumps",          "Time (min)",     "Heart Rate",
                             (m1, m2, m3) => (m1 * 0.012) + (m2 * 0.8) + (m3 * 0.05)),
        };

        public MultiCalcForm(List<string> activities)
        {
            selectedActivities = activities;
            Text               = "FitTrack — Enter Metrics & Calculate";
            // Form height scales with number of activities (capped at 800px)
            int formHeight     = Math.Min(800, 160 + activities.Count * 180);
            Size               = new Size(580, formHeight);
            StartPosition      = FormStartPosition.CenterParent;
            BackColor          = Theme.Background;
            FormBorderStyle    = FormBorderStyle.FixedDialog;
            MaximizeBox        = false;
            BuildUI();
        }

        /// <summary>
        /// Dynamically creates metric input rows for each selected activity.
        /// Uses a scrollable panel to handle large numbers of activities.
        /// </summary>
        void BuildUI()
        {
            Controls.Add(new Label
            {
                Text = "📊 Enter Metrics", ForeColor = Theme.Accent, Font = Theme.FontTitle,
                Left = 30, Top = 18, AutoSize = true, BackColor = Color.Transparent
            });
            Controls.Add(Theme.StyledLabel(
                $"Activities: {string.Join(", ", selectedActivities)}",
                30, 62, Theme.FontSmall, Theme.TextSec));

            // Scrollable panel holds all activity metric sections
            pnlScroll = new Panel
            {
                Left       = 20, Top = 88, Width = 540,
                Height     = Size.Height - 220,
                AutoScroll = true, BackColor = Theme.Background
            };
            Controls.Add(pnlScroll);

            int y = 0;
            metricBoxes.Clear();

            foreach (var act in selectedActivities)
            {
                var def = ActivityDefs[act];

                // Activity section header bar
                var hdr = new Panel { Left = 0, Top = y, Width = 520, Height = 34, BackColor = Theme.Card };
                hdr.Controls.Add(new Label
                {
                    Text      = $"  {Theme.GetActivityIcon(act)} {act}",  // FIX: use Theme.GetActivityIcon
                    Left      = 8, Top = 7, AutoSize = true,
                    ForeColor = Theme.Accent, Font = Theme.FontHeading, BackColor = Color.Transparent
                });
                pnlScroll.Controls.Add(hdr);
                y += 38;

                // Three metric input boxes in one row
                var box = new Panel { Left = 0, Top = y, Width = 520, Height = 120, BackColor = Theme.Surface };
                var tb1 = AddMetricField(box, def.m1, 10,  10);
                var tb2 = AddMetricField(box, def.m2, 180, 10);
                var tb3 = AddMetricField(box, def.m3, 350, 10);
                pnlScroll.Controls.Add(box);
                metricBoxes.Add((tb1, tb2, tb3));
                y += 128;
            }

            // Result display label
            lblResult = new Label
            {
                Left = 20, Top = Size.Height - 130, Width = 540, Height = 24,
                ForeColor = Theme.Warning, Font = Theme.FontHeading,
                BackColor = Color.Transparent, Text = ""
            };
            Controls.Add(lblResult);

            // Calculate button (shows result without saving)
            var btnCalc = Theme.StyledButton("🔥 CALCULATE", Theme.AccentDark);
            btnCalc.Left = 20; btnCalc.Top = Size.Height - 100; btnCalc.Width = 250;
            Controls.Add(btnCalc);
            btnCalc.Click += BtnCalc_Click;

            // Save button (calculates and persists to JSON)
            var btnSave = Theme.StyledButton("💾 SAVE ALL");
            btnSave.Left = 290; btnSave.Top = Size.Height - 100; btnSave.Width = 250;
            Controls.Add(btnSave);
            btnSave.Click += BtnSave_Click;
        }

        /// <summary>
        /// Adds a labelled metric input field to the given parent panel.
        /// </summary>
        /// <returns>The TextBox created for this metric.</returns>
        TextBox AddMetricField(Panel parent, string label, int x, int y)
        {
            parent.Controls.Add(new Label
            {
                Text = label, Left = x, Top = y, Width = 160, AutoSize = false,
                ForeColor = Theme.TextSec, Font = Theme.FontSmall, BackColor = Color.Transparent
            });
            var tb = Theme.StyledTextBox();
            tb.Left = x; tb.Top = y + 20; tb.Width = 158;
            parent.Controls.Add(tb);
            return tb;
        }

        /// <summary>
        /// Validates all metric input fields and calculates calories for each activity.
        /// All three metrics must be valid positive numbers.
        /// </summary>
        /// <param name="error">Error message if validation fails; otherwise null.</param>
        /// <returns>List of calorie values per activity, or null if validation failed.</returns>
        List<double> CalculateAll(out string error)
        {
            error = null;
            var results = new List<double>();

            // Loop through each selected activity
            for (int i = 0; i < selectedActivities.Count; i++)
            {
                var (tb1, tb2, tb3) = metricBoxes[i];

                // Parse each metric — all three must be valid positive numbers
                if (!double.TryParse(tb1.Text, out double m1) || m1 <= 0 ||
                    !double.TryParse(tb2.Text, out double m2) || m2 <= 0 ||
                    !double.TryParse(tb3.Text, out double m3) || m3 <= 0)
                {
                    error = $"✗ Please fill all fields for {selectedActivities[i]} with valid positive numbers.";
                    return null;
                }

                // Apply this activity's calorie formula from the dictionary
                double cal = ActivityDefs[selectedActivities[i]].calc(m1, m2, m3);
                results.Add(cal);
            }
            return results;
        }

        /// <summary>Calculates and displays total calories without saving.</summary>
        void BtnCalc_Click(object s, EventArgs e)
        {
            var results = CalculateAll(out string error);
            if (results == null)
            {
                lblResult.Text      = error;
                lblResult.ForeColor = Theme.Danger;
                return;
            }

            double total = results.Sum();
            var parts    = selectedActivities.Select((a, i) => $"{a}: {results[i]:F1}").ToList();
            lblResult.Text      = $"🔥 Total: {total:F1} kcal  ({string.Join("  |  ", parts)})";
            lblResult.ForeColor = Theme.Warning;
        }

        /// <summary>
        /// Calculates calories, creates ActivityEntry objects, saves to JSON,
        /// and fires the OnSaved callback to return to the Dashboard.
        /// </summary>
        void BtnSave_Click(object s, EventArgs e)
        {
            var results = CalculateAll(out string error);
            if (results == null)
            {
                lblResult.Text      = error;
                lblResult.ForeColor = Theme.Danger;
                return;
            }

            // Create and store an ActivityEntry for each selected activity
            for (int i = 0; i < selectedActivities.Count; i++)
            {
                var (tb1, tb2, tb3) = metricBoxes[i];
                double.TryParse(tb1.Text, out double m1);
                double.TryParse(tb2.Text, out double m2);
                double.TryParse(tb3.Text, out double m3);

                UserStore.CurrentUser.Activities.Add(new ActivityEntry
                {
                    ActivityName   = selectedActivities[i],
                    Metric1        = m1,
                    Metric2        = m2,
                    Metric3        = m3,
                    CaloriesBurned = results[i],
                    Date           = DateTime.Today
                });
            }

            UserStore.Save(); // Persist all new entries to disk

            double total = results.Sum();
            MessageBox.Show(
                $"✔ {selectedActivities.Count} activit{(selectedActivities.Count == 1 ? "y" : "ies")} saved!\n" +
                $"Total: {total:F1} kcal burned today.",
                "Saved", MessageBoxButtons.OK, MessageBoxIcon.Information);

            OnSaved?.Invoke(); // Navigate back to Dashboard
        }
    }

                    ForeColor = achieved ? Theme.Success : Theme.Danger,
                    BackColor = Color.Transparent
                });
                banner.Controls.Add(new Label
                {
                    Text      = achieved
                        ? $"Exceeded by {todayTotal - u.CalorieGoal:F0} kcal 💪"
                        : $"{u.CalorieGoal - todayTotal:F0} kcal more needed today.",
                    Left      = 66, Top = 46, AutoSize = true,
                    Font      = Theme.FontSmall, ForeColor = Theme.TextSec,
                    BackColor = Color.Transparent
                });
                Controls.Add(banner);
            }
        }
    }

    // ══════════════════════════════════════════════════════════
    //  FORM 8 — PROGRESS HISTORY
    // ══════════════════════════════════════════════════════════

    /// <summary>
    /// Shows all recorded activity history grouped by date (most recent first).
    /// Each day shows its total calories and a goal-achieved indicator.
    /// Uses LINQ GroupBy to aggregate activities by date.
    /// </summary>
    class HistoryForm : Form
    {
        public HistoryForm()
        {
            Text            = "FitTrack — Progress History";
            Size            = new Size(660, 660);
            StartPosition   = FormStartPosition.CenterParent;
            BackColor       = Theme.Background;
            FormBorderStyle = FormBorderStyle.FixedDialog;
            MaximizeBox     = false;
            BuildUI();
        }

        /// <summary>
        /// Builds the scrollable history list grouped by day.
        /// Uses LINQ GroupBy and OrderByDescending for clean data grouping.
        /// </summary>
        void BuildUI()
        {
            var u = UserStore.CurrentUser;

            Controls.Add(new Label
            {
                Text = "📅 Progress History", ForeColor = Theme.Accent, Font = Theme.FontTitle,
                Left = 30, Top = 20, AutoSize = true, BackColor = Color.Transparent
            });
            Controls.Add(Theme.StyledLabel(
                "Your calorie burn history — grouped by day", 30, 70, Theme.FontSmall, Theme.TextSec));

            // Group all activities by date, most recent first
            var grouped = u.Activities
                .GroupBy(a => a.Date.Date)
                .OrderByDescending(g => g.Key)
                .ToList();

            if (grouped.Count == 0)
            {
                Controls.Add(Theme.StyledLabel(
                    "No history yet. Start adding activities!", 30, 120, Theme.FontHeading, Theme.TextSec));
                return;
            }

            // Scrollable panel for the history list
            var pnlScroll = new Panel
            {
                Left = 30, Top = 100, Width = 590, Height = 500,
                AutoScroll = true, BackColor = Theme.Background
            };
            Controls.Add(pnlScroll);

            int y = 0;
            foreach (var group in grouped)
            {
                DateTime date      = group.Key;
                double dayTotal    = group.Sum(a => a.CaloriesBurned);
                bool isToday       = date == DateTime.Today;
                bool isYesterday   = date == DateTime.Today.AddDays(-1);
                bool goalReached   = u.CalorieGoal > 0 && dayTotal >= u.CalorieGoal;

                // Human-friendly day label
                string dayLabel = isToday     ? "Today"
                                : isYesterday ? "Yesterday"
                                : date.ToString("dddd");

                // Day header bar
                var hdrPanel = new Panel
                {
                    Left = 0, Top = y, Width = 570, Height = 42, BackColor = Theme.Card
                };

                hdrPanel.Controls.Add(new Label
                {
                    Text      = $"  📅 {dayLabel}  —  {date:dd MMM yyyy}",
                    Left      = 0, Top = 0, Width = 370, Height = 42,
                    ForeColor = isToday ? Theme.Accent : Theme.TextPrimary,
                    Font      = Theme.FontHeading,
                    TextAlign = ContentAlignment.MiddleLeft,
                    BackColor = Color.Transparent
                });

                hdrPanel.Controls.Add(new Label
                {
                    Text      = $"{dayTotal:F0} kcal  {(goalReached ? "✅" : "")}",
                    Left      = 370, Top = 0, Width = 190, Height = 42,
                    ForeColor = goalReached ? Theme.Success : Theme.Warning,
                    Font      = Theme.FontHeading,
                    TextAlign = ContentAlignment.MiddleRight,
                    BackColor = Color.Transparent
                });

                pnlScroll.Controls.Add(hdrPanel);
                y += 46;

                // Individual activity rows for this day
                foreach (var a in group.OrderBy(x => x.Date))
                {
                    var row = new Panel
                    {
                        Left = 10, Top = y, Width = 555, Height = 30, BackColor = Color.Transparent
                    };
                    row.Controls.Add(new Label
                    {
                        // FIX: Use Theme.GetActivityIcon() — no more duplicated GetIcon() method
                        Text      = $"  {Theme.GetActivityIcon(a.ActivityName)} {a.ActivityName}",
                        Left      = 0, Top = 5, Width = 200,
                        ForeColor = Theme.TextPrimary, Font = Theme.FontBody, BackColor = Color.Transparent
                    });
                    row.Controls.Add(new Label
                    {
                        Text      = $"{a.CaloriesBurned:F1} kcal",
                        Left      = 210, Top = 5, Width = 120,
                        ForeColor = Theme.TextSec, Font = Theme.FontBody, BackColor = Color.Transparent
                    });
                    pnlScroll.Controls.Add(row);
                    y += 32;
                }

                y += 10; // Spacer between day groups
            }

            // Summary footer
            double allTime  = u.Activities.Sum(a => a.CaloriesBurned);
            int totalDays   = grouped.Count;
            Controls.Add(Theme.StyledLabel(
                $"Total: {allTime:F0} kcal across {totalDays} day(s)",
                30, 608, Theme.FontSmall, Theme.TextSec));
        }
    }
